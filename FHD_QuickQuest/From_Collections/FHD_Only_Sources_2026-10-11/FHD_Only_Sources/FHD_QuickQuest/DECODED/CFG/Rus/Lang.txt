Script ^{
    FHDQuickQuest10 ^{
        Process ^{
            01=int state=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',1);
            02=int money=ShipMoney(Player());
            03=str form=CurrentForm();
            04=int inQuest=0;if(form=='PlanetQuest'||form=='PQuest'||form=='PQuest2')inQuest=1;
            05=if((ScriptItemActionType(t_OnEnteringForm)||ScriptItemActionType(t_OnReEnteringForm))&&inQuest&&state==0)
            06={
            07=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',1,1);
            08=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',2,money);
            09=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',3,0);
            10=exit;
            11=}
            12=if(ScriptItemActionType(t_OnLeavingForm)&&inQuest&&state==1)
            13={
            14=int baseMoney=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',2);
            15=int deltaMoney=money-baseMoney;
            16=int markerMagnitude=deltaMoney;if(markerMagnitude<0)markerMagnitude=-markerMagnitude;
            17=if(markerMagnitude>=850000000&&markerMagnitude<=1450000000)
            18={
            19=int markerIndex=(markerMagnitude-900000000)/50000000;
            20=if(markerIndex<0)markerIndex=0;if(markerIndex>10)markerIndex=10;
            21=int selectedPercent=10+markerIndex;
            22=int markerValue=900000000+markerIndex*50000000;
            23=int correctedMoney=money;if(deltaMoney>0)correctedMoney=money-markerValue;else correctedMoney=money+markerValue;
            24=ShipMoney(Player(),correctedMoney);
            25=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',1,2);
            26=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',2,correctedMoney);
            27=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',3,selectedPercent);
            28=}
            29=else
            30={
            31=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',1,0);
            32=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',3,0);
            33=}
            34=exit;
            35=}
            36=if(state==2)
            37={
            38=int lastMoney=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',2);
            39=int feePercent=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',3);
            40=if(feePercent<10||feePercent>20)feePercent=10;
            41=int rewardMoney=money-lastMoney;
            42=if(form=='Gov'&&rewardMoney>0)
            43={
            44=int rewardHundreds=rewardMoney/100;
            45=int rewardRest=rewardMoney-rewardHundreds*100;
            46=int johnFee=rewardHundreds*feePercent+(rewardRest*feePercent)/100;
            47=if(johnFee<1)johnFee=1;
            48=ShipMoney(Player(),money-johnFee);
            49=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',1,0);
            50=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',3,0);
            51=exit;
            52=}
            53=if(form=='Gov')
            54={
            55=if(ScriptItemActionType(t_OnLeavingForm))
            56={
            57=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',1,0);
            58=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',3,0);
            59=}
            60=exit;
            61=}
            62=if(ScriptItemActionType(t_OnStep)||ScriptItemActionType(t_OnEnteringForm)||ScriptItemActionType(t_OnReEnteringForm)||ScriptItemActionType(t_OnLeavingForm))
            63={
            64=ShipCustomShipInfoData(Player(),'FHDQuickQuestCore10',2,money);
            65=}
            66=}
        }
    }
}
